```c
typedef struct settingsStorage_t {

    uint32_t brightness;                 // 0-255
    uint32_t colorPalette;               // 调色板索引
    uint32_t enableGraph;                // 图表开关
    uint32_t graphPos;                   // 图表位置 0-3
    uint32_t graphSize;                  // 图表尺寸 0=小 1=大

    uint32_t enableMinValueDisplay;      // 最小值准星

    uint32_t useBlackFlashBang;          // 黑色闪屏
    uint32_t use4117Cursors;             // 4117 内置准星
    uint32_t __tail;                     // 0x1715600D
} settingsStorage_t;
```
